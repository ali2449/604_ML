📥 Data Loading & Preprocessing
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer

df = pd.read_csv('healthcare_dataset.csv')

# Feature engineering
df['Date of Admission'] = pd.to_datetime(df['Date of Admission'])
df['Discharge Date'] = pd.to_datetime(df['Discharge Date'])
df['Length of Stay'] = (df['Discharge Date'] - df['Date of Admission']).dt.days

# Define target
target = 'Test Results'
X = df.drop(columns=[target, 'Name', 'Doctor', 'Hospital', 'Date of Admission', 'Discharge Date'])
y = df[target]

# Preprocessing pipeline
categorical_cols = ['Gender', 'Blood Type', 'Medical Condition', 'Admission Type', 'Medication', 'Insurance Provider']
numerical_cols = ['Age', 'Billing Amount', 'Room Number', 'Length of Stay']

preprocessor = ColumnTransformer([
    ('num', StandardScaler(), numerical_cols),
    ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_cols)
])

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, stratify=y, random_state=42)
X_train_proc = preprocessor.fit_transform(X_train)
X_test_proc = preprocessor.transform(X_test)
🌲 Random Forest with GridSearch
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import GridSearchCV

rf = RandomForestClassifier(random_state=42)
param_grid = {
    'n_estimators': [100, 200],
    'max_depth': [10, 20, None],
    'min_samples_split': [2, 5]
}
grid_rf = GridSearchCV(rf, param_grid, cv=3, scoring='accuracy', n_jobs=-1)
grid_rf.fit(X_train_proc, y_train)
best_rf = grid_rf.best_estimator_
print("Best params:", grid_rf.best_params_)
⚡ XGBoost with Early Stopping
import xgboost as xgb
from sklearn.model_selection import train_test_split

# Split train into train/val for early stopping
X_train_part, X_val, y_train_part, y_val = train_test_split(X_train_proc, y_train, test_size=0.2, random_state=42)

xgb_model = xgb.XGBClassifier(eval_metric='mlogloss', random_state=42)
param_grid = {
    'learning_rate': [0.05, 0.1],
    'n_estimators': [100, 200],
    'max_depth': [3, 6],
    'subsample': [0.8, 1.0]
}
grid_xgb = GridSearchCV(xgb_model, param_grid, cv=3, scoring='accuracy')
grid_xgb.fit(X_train_part, y_train_part)
best_params = grid_xgb.best_params_

# Refit with early stopping
final_xgb = xgb.XGBClassifier(**best_params, eval_metric='mlogloss', early_stopping_rounds=10)
final_xgb.fit(X_train_part, y_train_part, eval_set=[(X_val, y_val)], verbose=False)
📈 Evaluation Metrics
from sklearn.metrics import accuracy_score, classification_report, roc_auc_score, confusion_matrix

y_pred = final_xgb.predict(X_test_proc)
print("Accuracy:", accuracy_score(y_test, y_pred))
print(classification_report(y_test, y_pred))

# ROC-AUC for multi-class
y_proba = final_xgb.predict_proba(X_test_proc)
roc_auc = roc_auc_score(y_test, y_proba, multi_class='ovr')
print("ROC-AUC:", roc_auc)
📚 Quick Theory: Random Forest vs XGBoost
🌲 Random Forest (Bagging)
• Parallel trees on bootstrapped data
• Random feature subsets per split
• Reduces variance
• OOB score for validation
• Robust to outliers
⚡ XGBoost (Boosting)
• Sequential trees correcting residuals
• Regularization (L1/L2)
• Reduces bias
• Early stopping, handles missing values
• Often higher accuracy with tuning
🔑 Key Insight: Random Forest is great out-of-the-box; XGBoost usually wins after careful hyperparameter tuning. Use both and compare!
